# Notrelix API — Complete Route Structure
# Backend: .NET 8 / ASP.NET Core / Clean Architecture + CQRS
# Version strategy: URL versioning /api/v1/
# Multi-tenant: workspace slug trong path + JWT claims

# ══════════════════════════════════════════════════════════════════
# BASE URL
# ══════════════════════════════════════════════════════════════════
# Production:  https://api.notrelix.com/api/v1
# Staging:     https://api-staging.notrelix.com/api/v1
# Local:       http://localhost:5000/api/v1

# ══════════════════════════════════════════════════════════════════
# MIDDLEWARE PIPELINE (áp dụng theo thứ tự cho mọi request)
# ══════════════════════════════════════════════════════════════════
# 1. RateLimitMiddleware       — sliding window per IP + per user
# 2. RequestLoggingMiddleware  — correlation-id, duration, status
# 3. ExceptionHandlingMiddleware — RFC 7807 Problem Details
# 4. AuthenticationMiddleware  — JWT Bearer verify
# 5. CurrentUserMiddleware     — extract userId/role → ICurrentUser
# 6. WorkspaceTenantMiddleware — resolve workspaceSlug → workspaceId (khi có)

# ══════════════════════════════════════════════════════════════════
# RESPONSE ENVELOPE
# ══════════════════════════════════════════════════════════════════
# Success:
# {
#   "success": true,
#   "data": { ... } | [...],
#   "meta": { "page": 1, "pageSize": 20, "total": 150, "hasMore": true }
# }
#
# Error (RFC 7807):
# {
#   "type": "https://api.notrelix.com/errors/not-found",
#   "title": "Resource not found",
#   "status": 404,
#   "detail": "Page with id 'xxx' was not found",
#   "traceId": "00-abc123-def456-00",
#   "errors": { "email": ["Email is already taken"] }
# }

# ══════════════════════════════════════════════════════════════════
# MODULE 1: AUTH — /api/v1/auth
# Middleware: RateLimit(strict) trên tất cả, không cần JWT
# ══════════════════════════════════════════════════════════════════

POST   /api/v1/auth/register
# Body: { name, email, password }
# 201: { user, accessToken } + Set-Cookie: refresh_token (httpOnly)
# 409: Email already exists

POST   /api/v1/auth/login
# Body: { email, password }
# 200: { user, accessToken } + Set-Cookie: refresh_token (httpOnly)
# 401: Invalid credentials

POST   /api/v1/auth/logout
# [AUTH REQUIRED] Cookie: refresh_token
# 204: revoke refresh token

POST   /api/v1/auth/refresh
# Cookie: refresh_token (httpOnly, tự động gửi)
# 200: { accessToken } — access token mới
# 401: Token expired / revoked

POST   /api/v1/auth/forgot-password
# Body: { email }
# 204: Luôn trả 204 (tránh email enumeration)

POST   /api/v1/auth/reset-password
# Body: { token, newPassword }
# 204: Password updated
# 400: Token invalid / expired

POST   /api/v1/auth/verify-email
# Body: { token }
# 204: Email verified

POST   /api/v1/auth/resend-verification
# [AUTH REQUIRED]
# 204: Resent

POST   /api/v1/auth/change-password
# [AUTH REQUIRED] Body: { currentPassword, newPassword }
# 204: Changed

# ── OAuth SSO ──────────────────────────────────────────────────
GET    /api/v1/auth/oauth/{provider}
# provider: google | github | microsoft
# 302: Redirect to provider OAuth URL

GET    /api/v1/auth/oauth/{provider}/callback
# Query: { code, state }
# 302: Redirect to frontend với accessToken hoặc error

POST   /api/v1/auth/oauth/{provider}/disconnect
# [AUTH REQUIRED]
# 204: OAuth account disconnected

# ── Sessions ──────────────────────────────────────────────────
GET    /api/v1/auth/sessions
# [AUTH REQUIRED]
# 200: [{ id, deviceInfo, ipAddress, createdAt, isCurrent }]

DELETE /api/v1/auth/sessions/{sessionId}
# [AUTH REQUIRED] — revoke session cụ thể
# 204

DELETE /api/v1/auth/sessions
# [AUTH REQUIRED] — revoke tất cả sessions trừ current
# 204

# ── 2FA (Phase 2) ─────────────────────────────────────────────
POST   /api/v1/auth/2fa/enable
POST   /api/v1/auth/2fa/verify
POST   /api/v1/auth/2fa/disable
GET    /api/v1/auth/2fa/backup-codes

# 204

# 200: Updated member

DELETE /api/v1/workspaces/{slug}/members/{userId}

# HTTP STATUS CODE CONVENTIONS
# ══════════════════════════════════════════════════════════════════
# 200 OK          — GET, PATCH thành công
# 201 Created     — POST tạo resource mới, body = resource + Location header
# 202 Accepted    — async job queued (calendar sync, email...)
# 204 No Content  — DELETE, POST action (archive/restore) thành công
# 400 Bad Request — validation failed (kèm errors field)
# 401 Unauthorized — không có token hoặc token invalid
# 403 Forbidden   — có token nhưng không đủ quyền
# 404 Not Found   — resource không tồn tại
# 409 Conflict    — duplicate (email, slug) hoặc version conflict (block)
# 410 Gone        — resource đã bị xóa vĩnh viễn (invitation expired)
# 422 Unprocessable — business rule violation (không phải validation)
# 429 Too Many Requests — rate limit exceeded + Retry-After header
# 500 Internal Server Error — unexpected error (không expose stack trace)
